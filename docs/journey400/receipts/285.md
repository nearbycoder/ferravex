## Consumer recipe: both dimensional physics authoring boundaries

This complete first-party recipe is synthetic native input/CPU painting. It
creates candidate definitions with the **existing physics validators**, without
adding a physics model/controller to the renderer. The unit cube/rectangle has
area/volume one, so its authored density equals mass in this specific fixture.
General selected-body mass editing needs the actual collider geometry and
`MassProperties`; there is no generic `set_mass` API. Gravity here is explicitly
the body's dimensionless gravity scale, checked by `try_gravity_scale`; it is
not mislabeled as a checked world-acceleration setter. World gravity remains an
existing finite-vector authoring boundary. Static/sliding friction relations
remain `PhysicsMaterial::validate`'s authority.

The same snapshot is used by all three controls and the callback. Candidate
construction is pure. Only an accepted result applies values after finish.
Rejecting sliding friction from 0.5 to 0.75 against static friction 0.6 is a true
no-op on both the authored values and previously constructed physics objects.
It also checks complete display/units, enabled button inspection and the last
registered action's exact painted/hit rectangle. No lab source was edited.

```rust,check crate-type=bin
use ferravex_play::ui::theme::{ThemeMode, UiTheme};
use ferravex_physics::{PhysicsMaterial, d2, d3};
use ferravex_render::{Vec2, input::{host_frame, touch, touch_event},
    ui::{Ui, Viewport, inspector::*}};

fn main() {
    macro_rules! dimension {
        ($module:ident, $collider:expr) => {{
            let make = |mass: f32, gravity: f32, sliding: f32| {
                let material = PhysicsMaterial { sliding_friction: sliding,
                    static_friction: 0.6, ..PhysicsMaterial::default() };
                material.validate().map_err(|_| "friction relation")?;
                let collider = $collider.try_density(mass)
                    .map_err(|_| "density")?.material(material)
                    .try_build().map_err(|_| "collider")?;
                let body = $module::RigidBodyBuilder::dynamic()
                    .try_gravity_scale(gravity).map_err(|_| "gravity")?
                    .try_build_with_colliders(std::slice::from_ref(&collider))
                    .map_err(|_| "body/mass")?;
                body.validate_definition(std::slice::from_ref(&collider))
                    .map_err(|_| "definition")?;
                Ok::<_, &'static str>((body, collider))
            };
            let mut authored = (1.0_f32, 1.0_f32, 0.5_f32);
            let mut definition = make(authored.0, authored.1, authored.2).unwrap();
            let mut ui = Ui::new();
            ui.set_inspection_enabled(true);
            let viewport = Viewport::responsive([320., 700.], [0.; 4]).unwrap();
            // Real journal press/release: MASS+, GRAVITY+, FRICTION+ (rejected).
            for (action, expected) in [(3, (1.25, 1., 0.5)),
                (6, (1.25, 1.25, 0.5)), (9, (1.25, 1.25, 0.5))] {
                let snapshot = authored;
                let displays = [format!("{:.2}", snapshot.0),
                    format!("{:.2}", snapshot.1), format!("{:.2}", snapshot.2)];
                let fields = std::array::from_fn::<_, 3, _>(|i| InspectorField {
                    id: 1 + i as u64 * 3,
                    label: ["MASS", "GRAVITY SCALE", "SLIDING FRICTION"][i],
                    display: &displays[i], unit: ["kg", "x", ""][i],
                    state: FieldState::Enabled,
                    control: InspectorControl::Stepper {
                        value: [snapshot.0, snapshot.1, snapshot.2][i],
                        min: [0.1, -10., 0.][i], max: [100., 10., 1000.][i],
                        step: 0.25, policy: LimitPolicy::Reject,
                        decrease: 2 + i as u64 * 3, increase: 3 + i as u64 * 3,
                    },
                });
                let layout = InspectorLayout::new(viewport.bounds(), &fields,
                    UiTheme::preset(ThemeMode::Normal)).unwrap();
                let rect = layout.control_rect(action).unwrap();
                for phase in [touch::Phase::Start, touch::Phase::End] {
                    touch_event(touch::Event { id: action, phase,
                        position: Vec2::new(rect.min[0] + 2., rect.min[1] + 2.) });
                }
                let proposed = |change: ValueChange| {
                    let FieldValue::Number(n) = change.value else { panic!("numeric fixture") };
                    match change.field {
                        1 => (n, snapshot.1, snapshot.2),
                        4 => (snapshot.0, n, snapshot.2),
                        7 => (snapshot.0, snapshot.1, n),
                        _ => panic!("stable field ID"),
                    }
                };
                let before_density = definition.1.density();
                let before_material = definition.1.material();
                host_frame(0., 320, 700, || {
                    ui.begin(viewport);
                    let outcome = layout.finish(&mut ui, true, |change| {
                        let next = proposed(change);
                        make(next.0, next.1, next.2).map(|_| ())
                    }).unwrap();
                    assert_eq!(authored, snapshot);
                    if let InspectorOutcome::Accepted(change) = outcome {
                        let next = proposed(change);
                        definition = make(next.0, next.1, next.2).unwrap();
                        authored = next; // Application is outside speculative UI/validation.
                    } else {
                        assert_eq!(outcome, InspectorOutcome::Invalid { field: 7,
                            reason: InvalidEdit::Validation("friction relation") });
                        assert_eq!(definition.1.density(), before_density);
                        assert_eq!(definition.1.material(), before_material);
                    }
                    let inspected = ui.inspect().unwrap().find(|e| e.id == Some(action)).unwrap();
                    assert_eq!(inspected.bounds, rect);
                    assert_eq!(inspected.visible_bounds, Some(rect));
                    assert!(inspected.enabled && inspected.interaction_registered);
                    assert!(ui.interaction().pointer_consumed);
                    ui.paint();
                });
                assert_eq!(authored, expected);
            }
            // Definition is still accepted by the actual dimensional physics world.
            let mut world = $module::PhysicsWorld::new(Default::default());
            let handle = world.insert_compound(definition.0, [definition.1]).unwrap();
            assert!((world.bodies[handle].mass() - 1.25).abs() < 0.00001);
            ui.cancel(); // Close/restart explicitly cancels any owned gesture.
        }};
    }
    dimension!(d2, d2::ColliderBuilder::try_cuboid(0.5, 0.5).unwrap());
    dimension!(d3, d3::ColliderBuilder::try_cuboid(0.5, 0.5, 0.5).unwrap());
}
```

## Concrete behavior and compatibility

The layout borrows fields, display copy and choice slices, and retains only fixed
geometry arrays and a copied explicit `UiTheme`. Field IDs and control IDs must
be stable and distinct across the entire supplied inspector. Field IDs key
results/row queries; only control IDs register buttons. Duplicate control IDs
against prior UI are rejected by the actual `Ui` transaction. Unknown geometry
queries return None. There is no global theme, hidden field state, arbitrary
text-entry framework, physics controller, new input routing or renderer.

All source lengths are bounded before scanning labels/choices. Limits are 16
fields, eight choices per choice field, 128 control geometry slots and 8192
combined borrowed UTF-8 bytes, including all alternatives, values, units and
status messages. Generated prefixes have fixed additional stack capacity. No
String/Vec allocation or caller-model mutation is performed by this composer.
The existing command/widget/text/vertex/clip budgets remain authoritative;
passing the source/layout bound does not guarantee submission capacity.

Both preset themes use label 16/body/control 18 logical units. Author-supplied
validated type/spacing tokens are honored; no text is shrunk. Text is measured
with the actual shared WrappedText/bitmap metrics, retaining complete labels,
value plus units, validation/availability messages and current toggle/choice
state. Choice/toggle labels must be nonblank single lines; field copy may wrap
and contain newlines. The renderer's existing Unicode `?` substitution remains.
Numeric formatting/precision/units are caller-owned; display is never parsed.

Every row reserves label, value, all complete control rows and status copy.
Control rows wrap using measured widths, at least 44 logical units high; tiny
symbol widths currently reserve 45 units including rounding slack. Toggles
reserve their larger ON/OFF label width. Geometry is canonicalized to f32
endpoints and checked again at actual painter dimensions. Too narrow/short,
nonfinite or unrepresentable bounds return InvalidLayout before insertion;
there is no silent omission, tiny text, scroll state or fallback insertion.
Caller chooses a larger/dedicated view or fewer fields plus an explicit readable
fallback. Bounds must be inside safe/ancestor visible regions. Inherited clips
are preserved, and a fully clipped inspector cannot acquire a press. Inspection
reports the exact complete labels/button bounds/visible intersections.

Enabled fields alone may propose changes. Disabled, unavailable and invalid
fields retain distinct `X DISABLED:`, `X UNAVAILABLE:` and `! INVALID:` contextual
copy and inert buttons. Invalid/unavailable snapshots may carry an unknown
numeric value (including NaN) for display; numeric limits/step must always be
finite and coherent, and enabled values must be finite/in range. Disabled
choices own/block gestures through existing button behavior but never propose.
Selected choice and unchanged clamped endpoint activations return Idle.

Steppers require min <= max and finite step > 0. The default is deliberately not
inferred: each field explicitly chooses Reject or Clamp. Reject reports an
out-of-range candidate; Clamp clamps only a finite arithmetic result. Arithmetic
overflow/nonfinite results always return NumericLimit, including with Clamp.
A step rounded away in f32 returns NoNumericProgress. No silent clamp, wrap,
auto-repeat, precision formatter or domain constraint is installed.

`finish`/`finish_player` internally call `Ui::compose`, then the corresponding
existing input finish, then resolve at most one current accepted activation.
No proposal or callback runs inside speculative composition. UI submission error
rolls back the entire inspector and does not finish input; caller must finish
restored prior UI or a fallback. A pure validator runs exactly once only for a
real candidate change; its generic error is returned without copying/truncating
it. It can construct a candidate body/material definition, as above. Arbitrary
callback side effects cannot be made transactional; do not mutate the model in
that callback. Apply Accepted once outside the UI, then rebuild the snapshot.
There is no speculative nested submission API capable of escaping a value before
an enclosing transaction later fails. Submit other widgets first and call this
finish once, then paint explicitly. The existing last-activation aggregate may
collapse multiple gestures per host frame; this is not a new event queue.

Existing focus, Tab/arrows, Enter, selected-pad input, remapped global confirm,
release capture, disabled blocking and cancellation are reused unchanged.
Holding a key/gesture does not synthesize a new change per render. `begin` removes
old layout; explicit `Ui::cancel` closes/resets gestures. Continue input routing
for terminal tails after closing, including with an empty UI/disabled finish.
The inspector does not advance/pause simulation, consume gameplay edges, change
motion/flash settings, manage audio or authoritatively apply edits. Local tools
must keep network/replay progression intact. UI ownership must gate fresh scene
pointer gestures using `Ui::interaction` after finish; queued gameplay edges
still belong to actual ticks.

Theme Canvas/Surface/Text/MutedText/DisabledText/Danger/Focus tokens and current
type/spacing metrics are used only through accepted Ui arguments. Ordinary
retained buttons keep their existing two-unit focus stroke and static `[X]`/
`[ ]` selection cue. This is direct composition, **not a themed CardLayout**;
task277's fixed card style/API was not changed or claimed to consume tokens.
No new CardStyle overload or shared UI event behavior is required.

## Exact additive public API

All names are in `ferravex_render::ui::inspector`. Signature excerpts below are
not a second standalone program. Public data enums/structs are Clone+Copy+Debug;
LimitPolicy/FieldState additionally PartialEq+Eq, InspectorChoice/Control/Field
have no comparison contract, FieldValue/ValueChange/InspectorOutcome have
PartialEq, and InvalidEdit additionally Eq (subject to generic bounds).
InspectorLayout is Debug with private borrowed/fixed-storage fields.

```text
pub const MAX_INSPECTOR_FIELDS: usize = 16;
pub const MAX_INSPECTOR_CHOICES: usize = 8;
pub const MAX_INSPECTOR_BYTES: usize = 8192;
pub enum LimitPolicy { Reject, Clamp }
pub enum FieldState<'a> {
    Enabled, Disabled(&'a str), Unavailable(&'a str), Invalid(&'a str),
}
pub struct InspectorChoice<'a> {
    pub id: WidgetId, pub label: &'a str, pub enabled: bool,
}
pub enum InspectorControl<'a> {
    Stepper { value: f32, min: f32, max: f32, step: f32,
              policy: LimitPolicy, decrease: WidgetId, increase: WidgetId },
    Toggle { value: bool, id: WidgetId, on: &'a str, off: &'a str },
    Choice { selected: usize, choices: &'a [InspectorChoice<'a>] },
}
pub struct InspectorField<'a> {
    pub id: WidgetId, pub label: &'a str, pub display: &'a str,
    pub unit: &'a str, pub state: FieldState<'a>,
    pub control: InspectorControl<'a>,
}
pub enum FieldValue { Number(f32), Toggle(bool), Choice(usize) }
pub struct ValueChange {
    pub field: WidgetId, pub previous: FieldValue, pub value: FieldValue,
}
pub enum InvalidEdit<E> { NumericLimit, NoNumericProgress, Validation(E) }
pub enum InspectorOutcome<E> {
    Idle, Accepted(ValueChange), Invalid { field: WidgetId, reason: InvalidEdit<E> },
}
pub struct InspectorLayout<'a> { /* private fields */ }
impl<'a> InspectorLayout<'a> {
    pub fn new(bounds: UiRect, fields: &'a [InspectorField<'a>], theme: UiTheme)
        -> Result<Self, UiError>;
    pub fn bounds(&self) -> UiRect;
    pub fn field_rect(&self, id: WidgetId) -> Option<UiRect>;
    pub fn control_rect(&self, id: WidgetId) -> Option<UiRect>;
    pub fn finish<E>(&self, ui: &mut Ui, enabled: bool,
        validate: impl FnOnce(ValueChange) -> Result<(), E>)
        -> Result<InspectorOutcome<E>, UiError>;
    pub fn finish_player<E>(&self, ui: &mut Ui,
        player: &mut crate::input::PlayerInput, enabled: bool,
        validate: impl FnOnce(ValueChange) -> Result<(), E>)
        -> Result<InspectorOutcome<E>, UiError>;
}
```
